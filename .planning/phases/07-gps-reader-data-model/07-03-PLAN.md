    Refactor GPSReader so get_location() is an instant cache read and poll() does the I/O.
    This is a non-breaking refactor — tests in test_gps_reader.py that called get_location()
    for fetch behavior should be updated to call poll() instead.

    New GPSReader shape:
    ```python
    def __init__(self) -> None:
        if not _WINSDK_AVAILABLE:
            self.status: str = "unavailable"
        else:
            self.status = "no_fix"
        self._cache: tuple[float | None, float | None, float | None] = (None, None, None)

    def get_location(self) -> tuple[float | None, float | None, float | None]:
        """Return last cached GPS fix. Never raises. Background thread keeps cache fresh."""
        return self._cache

    def poll(self) -> None:
        """Fetch from Windows Location API and update cache. Called by background thread at ~1 Hz."""
        if not _WINSDK_AVAILABLE:
            return
        try:
            lat, lon, speed = asyncio.run(self._fetch())
            if lat is None or lon is None:
                self.status = "no_fix"
                self._cache = (None, None, None)
            else:
                self.status = "ok"
                self._cache = (lat, lon, speed)
        except PermissionError:
            self.status = "denied"
            self._cache = (None, None, None)
        except Exception as exc:
            logger.debug("GPS poll failed: %s", exc)
            self.status = "no_fix"
            self._cache = (None, None, None)
    ```

    Update tests/test_gps_reader.py: rename all calls from get_location() to poll()
    in the tests that test the fetch behavior. Add a new test:
    test_get_location_returns_cache — calls poll() with mocked _fetch, then verifies
    get_location() returns the cached result.

       ```python
       def _gps_poll_loop(gps_reader, stop_event) -> None:
           """Background thread: call gps_reader.poll() at ~1 Hz until stop_event set."""
           while not stop_event.is_set():
               gps_reader.poll()
               stop_event.wait(timeout=1.0)
       ```

    3. In main(), immediately after setup_logging() and before CSVLogger construction:
       ```python
       # GPS: initialize reader, do one poll for startup status, start background thread
       gps_reader = GPSReader() if GPSReader is not None else None
       gps_stop = threading.Event()
       if gps_reader is not None:
           gps_reader.poll()  # synchronous poll at startup to establish initial status
           gps_thread = threading.Thread(
               target=_gps_poll_loop,
               args=(gps_reader, gps_stop),
               daemon=True,
               name="gps-thread",
           )
           gps_thread.start()
           if gps_reader.status == "denied":
               print("GPS: location access denied — enable in Windows Settings › Privacy › Location")
           elif gps_reader.status == "ok":
       ```
