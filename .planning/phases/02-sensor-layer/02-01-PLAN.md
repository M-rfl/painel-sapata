def to_signed16(value: int) -> int:
    """Convert unsigned 16-bit Modbus register value to signed Python int.
    Modbus registers are unsigned 16-bit (0-65535). Apply to EVERY register
    before any scale factor. raw=32768 -> -32768, raw=65535 -> -1.
    """
    return value if value < 32768 else value - 65536

                    temp_raw=2500, disp_raw=0, freq_raw=10):
    """19 registers matching burst layout:

    [12]    0x40        Temperature
    [13-15] 0x41-0x43  Displacement X/Y/Z

        ax_raw, ay_raw, az_raw,                    # offsets 0-2
